# ---- добавить в Makefile (после блока драйверов) ----
USER_CFLAGS := -std=gnu11 -O2 -ffreestanding -fno-pic -fno-pie -fno-stack-protector \
               -mno-red-zone -mgeneral-regs-only -fno-asynchronous-unwind-tables \
               -fno-tree-loop-distribute-patterns $(COMMON_WARN) -I system/init
USER_PROGS  := init hello
USER_ELFS   := $(patsubst %,$(BUILD)/user/%.elf,$(USER_PROGS))
INITRD      := $(BUILD)/initrd.img

$(BUILD)/user/%.o: system/init/%.c system/init/syscall.h
	@mkdir -p $(dir $@)
	$(CC) $(USER_CFLAGS) -c $< -o $@

$(BUILD)/user/%.elf: $(BUILD)/user/%.o system/init/user.ld
	$(LD) -nostdlib -static -z max-page-size=0x1000 -z noexecstack -T system/init/user.ld -o $@ $<

$(INITRD): $(USER_ELFS) system/init/init.conf tools/mkinitrd.py
	python3 tools/mkinitrd.py $@ $(foreach p,$(USER_PROGS),$(p)=$(BUILD)/user/$(p).elf) init.conf=system/init/init.conf

# ---- заменить правило image и добавить initrd в .PHONY не нужно ----
image: $(BOOT_EFI) $(KERNEL_ELF) $(INITRD)
	@mkdir -p $(ESP)/EFI/BOOT $(ESP)/EFI/MIGA
	cp $(BOOT_EFI) $(ESP)/EFI/BOOT/BOOTX64.EFI
	cp $(KERNEL_ELF) $(ESP)/EFI/MIGA/kernel.elf
	cp $(INITRD) $(ESP)/EFI/MIGA/initrd.img
	@echo "ESP готова: $(ESP)"
